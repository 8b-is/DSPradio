     let client = BrokerClient::connect(None).await?;  // Auto-discover via

     callback).await?;                                                          
                                                                                
     while let Some(samples) = lease.next().await {                             
         if let Some(warning) = lease.check_preempt() {                         
             // Save state, acknowledge checkpoint

     Handoff Sequence (Critical Path)                                           
                                                                                
     T+0ms    Client B requests device (Priority: High)                         
              Current: Client A (Priority: Normal)

              Client B starts receiving I/Q
